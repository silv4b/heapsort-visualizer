Algoritmo em Execução

procedimento heapsort(A):
  n ← tamanho(A), ordenados ← []
  // 1. Transforma o array em Heap O(n)
  construirHeap(A, n)
  // 2. Extrai raiz sucessivamente na ordem
  enquanto heapSize > 0:
    raiz ← extrairRaiz(A)  // Vai para o array ordenado
    ordenados.anexar(raiz) // Na ordem exata (Max=Decrescente, Min=Crescente)
    heapify(A, heapSize, 0) // Restaura heap
            
procedimento heapify(A, n, i):
  alvo ← i, esq ← 2*i + 1, dir ← 2*i + 2
  // Compara com filho esquerdo
  se esq < n e A[esq] melhorQue A[alvo]: alvo ← esq
  // Compara com filho direito
  se dir < n e A[dir] melhorQue A[alvo]: alvo ← dir
  // Se o melhor não for a raiz, troca e propaga
  se alvo ≠ i:
    trocar(A[i], A[alvo])
    heapify(A, n, alvo)
            
procedimento inserir(A, valor):
  A.anexar(valor)
  i ← tamanho(A) - 1
  // Sobe o elemento enquanto violar o heap
  enquanto i > 0:
    pai ← ⌊(i - 1) / 2⌋
    se A[i] violaHeap A[pai]:
      trocar(A[i], A[pai])
      i ← pai
    senão: interromper
